<script>
(() => {
  window.bench = { done: false, status: "loading", phase: "module-load" };
  const publishFailure = (value, phase) => {
    if (window.bench?.done === true) return;
    let message;
    try { message = String(value?.message || value || "page module failed"); }
    catch { message = "page module failed with an unreadable error"; }
    window.bench = { ...window.bench, done: true, status: "failed", phase: window.bench?.phase || phase, error: message.slice(0, 1024) };
  };
  window.addEventListener("error", (event) => {
    const target = event.target;
    if (target !== window) {
      if (target?.tagName === "SCRIPT" && target.type === "module") publishFailure(`Module script failed to load: ${target.getAttribute("src") || "inline module"}`, "module-load");
      return;
    }
    if (event instanceof ErrorEvent) publishFailure(event.message || event.error, "module-runtime");
  }, true);
  window.addEventListener("unhandledrejection", (event) => publishFailure(event.reason, "module-runtime"));
})();
</script>
