const median = (values) => {
  const sorted = [...values].sort((a, b) => a - b);
  const i = Math.floor(sorted.length / 2);
  return sorted.length % 2 ? sorted[i] : (sorted[i - 1] + sorted[i]) / 2;
};
async function sha256(value) {
  return Array.from(new Uint8Array(await crypto.subtle.digest("SHA-256", value)), (v) => v.toString(16).padStart(2, "0")).join("");
}

// Called only after the explicit run query, never by idle or source smoke.
function fixtureData(id) {
  let state = seeds[id];
  const next = () => state = (Math.imul(state, 1664525) + 1013904223) >>> 0;
  const X = new Float32Array(512 * 1024);
  const W = new Int8Array(5248 * 1024);
  const S = new Float32Array(5248 * 32);
  const half = id === "cancellation" ? 16 : 32;
  for (let blockIndex = 0; blockIndex < 512 * 32; blockIndex++) {
    const start = blockIndex * 32;
    for (let k = 0; k < half; k++) {
      const draw = next();
      let value = id === "integer" || id === "dyadic" ? draw % 255 - 127 : Math.fround(((draw >>> 8) % 131075 - 65537) / 65537);
      if (k === 0) value = id === "integer" || id === "dyadic" ? 127 : 1;
      X[start + k] = id === "dyadic" ? value / 256 : value;
      if (half === 16) X[start + k + 16] = Math.fround(-value * (1 - 2 ** -22));
    }
  }
  for (let blockIndex = 0; blockIndex < 5248 * 32; blockIndex++) {
    const start = blockIndex * 32;
    for (let k = 0; k < half; k++) {
      const value = next() % 256 - 128;
      W[start + k] = value;
      if (half === 16) W[start + k + 16] = value;
    }
  }
  for (let i = 0; i < S.length; i++) {
    const draw = next();
    S[i] = id === "integer" ? 1 : id === "dyadic" ? 1 / 16 : 2 ** (-7 - draw % 3);
  }
  return { X, W, S };
}

        }

    }
    phase("fixed-diagnostic-samples");
    for (let index = 0; index < 9; index++) {
      const row = { block: "diagnostic", index, kernel: "candidate" };
      result.timing.samples.push(row); await sample(row);
    }
    const samples = result.timing.samples;
    result.timing.summary = { sampleCount: 9, invocationsPerSample: repetitions,
      medianStageTotalsMs: { quantize: median(samples.map((row) => row.stageTotalsMs.quantize)),
        integerMatmul: median(samples.map((row) => row.stageTotalsMs.integerMatmul)), enclosingTotal: median(samples.map((row) => row.gpuMs)) },
      medianPerInvocationMs: { quantize: median(samples.map((row) => row.stageTotalsMs.quantize / repetitions)),
        integerMatmul: median(samples.map((row) => row.stageTotalsMs.integerMatmul / repetitions)), enclosingTotal: median(samples.map((row) => row.perInvocationMs)) },
      zeroDurationPasses: { quantize: samples.reduce((sum, row) => sum + row.zeroDurationPasses.quantize, 0),
        integerMatmul: samples.reduce((sum, row) => sum + row.zeroDurationPasses.integerMatmul, 0) },
      negativeInterPassGaps: samples.reduce((sum, row) => sum + row.negativeInterPassGaps, 0),
      meaning: diagnostic.instrumentationLimitation };
    assert(result.fixtures.length === 4 && result.warmups.length === 3 && samples.length === 9, "Incomplete fixed diagnostic protocol");
    healthy(); result.status = "done";
  } catch (error) {
    const text = String(error?.stack || error);
    result.error = text.slice(0, 8192); result.errorTruncated = text.length > 8192;
    recordError({ phase: result.phase, message: text }); result.status = "error";
  } finally {
    if (device) {
      phase("cleanup"); cleaning = true;
      for (const buffer of [...buffers]) {
        try { destroyBuffer(buffer); } catch (error) { recordError({ phase: "cleanup", message: String(error) }); result.status = "error"; }
      }
      for (const queries of querySets) {
        try { queries.destroy(); result.cleanup.querySetsDestroyed++; } catch (error) { recordError({ phase: "cleanup", message: String(error) }); result.status = "error"; }
      }
      querySets.clear();
      try { device.destroy(); result.cleanup.deviceDestroyed = true; } catch (error) { recordError({ phase: "cleanup", message: String(error) }); result.status = "error"; }
      if (result.cleanup.buffersCreated !== result.cleanup.buffersDestroyed || result.cleanup.querySetsCreated !== result.cleanup.querySetsDestroyed || result.cleanup.mapsOpen !== 0) {
        recordError({ phase: "cleanup", message: "Resource cleanup counts differ" }); result.status = "error";
      }
    }
    if (physical) result.phase = result.status === "done" ? "complete" : "failed";
    if (physical || smoke) result.done = true;
    document.getElementById("status").textContent = result.error ?? result.status;
  }
}
