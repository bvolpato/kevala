import assert from "node:assert/strict";
import { readFileSync } from "node:fs";
import test from "node:test";
import { validateOperatorResult } from "./integer-matrix-operator-result.mjs";

// Metadata-only controller fixtures. No page/helper imports, tensor generation or GPU work.
const protocol = JSON.parse(readFileSync(new URL("./laya-integer-matrix-protocol.json", import.meta.url), "utf8"));
const contract = JSON.parse(readFileSync(new URL("./integer-matrix-operator-contract.json", import.meta.url), "utf8"));
const words = contract.correctness.requiredStatusWords;
const orders = contract.timing.orders;
const validation = () => ({ words: [...words], partUnchanged: true, passed: true,
  positiveCoverage: { outputs: 2686976, quantValues: 524288, quantBlocks: 16384, canaries: 512 } });
function sample(block, quartet, position, logicalLane, kernel) {
  const elapsed = kernel === "candidate" ? 48000000 : 32000000;
  return { block, quartet, position, logicalLane, kernel, invocations: 16, passes: kernel === "candidate" ? 32 : 16,
    dispatches: { baseline: kernel === "baseline" ? 16 : 0, quantize: kernel === "candidate" ? 16 : 0, integerMatmul: kernel === "candidate" ? 16 : 0, reduce: 0 },
    timestampNs: { begin: "1000", end: String(1000 + elapsed), elapsed: String(elapsed) }, gpuMs: elapsed / 1e6, perInvocationMs: elapsed / 1e6 / 16, validation: validation() };
}
function metadata() {
  const sourceIdentities = {};
  for (const [name, bytes, sha256] of [
    ["helper", 12993, "70e56d7d3ba2053fb46f84573f95647366e694f86950c3f3f1ee1bedc71460eb"],
    ["protocol", 19823, "cf1a5470780504b3aa20147f73935f96da08faf0d7d25a177545ec92512f3a39"],
    ["baseline", 13558, "0710c1bec14afcf918a2ac6785bc55c6f0ebf6580531ec491621f14a02cbe89c"],
    ["reducer", 1546, "0c2b71d87205b4fdae5d0ac63131545ff8f39a36cb5558b0fe80700562e40c3a"],
  ]) sourceIdentities[name] = { bytes, sha256 };
  const features = ["subgroups", "chromium-experimental-subgroup-matrix", "timestamp-query"];
  const result = { done: true, status: "done", phase: "complete", error: null, errors: [], sourceIdentities,
    compilation: Array.from({ length: 7 }, (_, index) => ({ name: `metadata-only-${index}`, diagnostics: { totalMessages: 0, totalErrors: 0, truncated: false, messages: [] } })),
    protocol: { schema: protocol.schema, shape: { M: 512, N: 5248, K: 1024, block: 32 }, repetitions: 16, orders,
      warmupsPerLane: 3, AAQuartets: 9, pairedQuartets: 9, reducerDispatches: 0, modelOrWasmLoads: 0, promotion: false },
    capabilities: { config: { componentType: "i8", resultComponentType: "i32", M: 16, N: 16, K: 32 }, requiredLimits: {},
      requestedFeatures: features, features, shaderF16Requested: false, subgroupMinSize: 32, subgroupMaxSize: 32,
      adapter: { vendor: "nvidia", architecture: "blackwell", description: "metadata-only test" },
      deviceLimits: { maxComputeWorkgroupStorageSize: 16384, maxComputeInvocationsPerWorkgroup: 256, maxComputeWorkgroupSizeX: 256, maxStorageBufferBindingSize: 2 ** 27 } },
    adapterIdentity: { nvidia: true, blackwell: true, nonfallback: true, isFallbackAdapter: false, fallbackFlags: { adapterInfo: false, adapter: null } },
    packingGate: { passed: true, outputsCompared: 256, mismatches: 0, guardsCompared: 128, guardFailures: 0, dispatches: 1,
      positive: 106, negative: 150, zero: 0, actual: protocol.prerequisiteGate.fixture.expected.values,
      actualSha256: "498117f912a5f1a4a3a2f6cd8bda7bacd741f38d389ea754e842b8fa463a58b3", expectedSha256: "498117f912a5f1a4a3a2f6cd8bda7bacd741f38d389ea754e842b8fa463a58b3" },
    fixtures: protocol.fixtures.map((fixture) => ({ id: fixture.id, seed: fixture.seed, shape: [512, 5248, 1024], exact: ["integer", "dyadic"].includes(fixture.id), status: "pass",
      inputIdentities: { X: { bytes: 2097152, sha256: "0".repeat(64) }, W: { bytes: 5373952, sha256: "1".repeat(64) }, S: { bytes: 671744, sha256: "2".repeat(64) } },
      dispatches: { baseline: 1, quantize: 1, integerMatmul: 1, reduce: 0 }, validation: validation() })),
    warmups: Array.from({ length: 6 }, (_, index) => sample("warmup", null, Math.floor(index / 2), index % 2 ? "B" : "A", index % 2 ? "candidate" : "baseline")),
    timing: { aa: [], paired: [], summary: { medianGain: -0.5, maxAASpread: 0, byOrder: { ABBA: -0.5, BAAB: -0.5 }, winningQuartets: 0,
      criteria: { medianAtLeastFivePercent: false, bothOrdersPositive: false, exceedsAASpread: false }, accepted: false } },
    cleanup: { buffersCreated: 5, buffersDestroyed: 5, querySetsCreated: 1, querySetsDestroyed: 1, mapsOpen: 0, deviceDestroyed: true } };
  for (const block of ["aa", "paired"]) result.timing[block] = orders.map((order, index) => ({ index, order,
    rows: [...order].map((lane, position) => sample(block, index, position, lane, block === "aa" || lane === "A" ? "baseline" : "candidate")),
    baselineMs: 2, comparisonMs: block === "aa" ? 2 : 3, gain: block === "aa" ? 0 : -0.5 }));
  return result;
}

test("metadata contract accepts a complete slower result without inventing a performance win", () => {
  const result = validateOperatorResult(metadata());
  assert.equal(result.timedRows, 72); assert.equal(result.performanceAccepted, false); assert.equal(result.medianGain, -0.5);
});
test("metadata contract rejects one dropped paired timing row", () => {
  const page = metadata(); page.timing.paired[8].rows.pop();
  assert.throws(() => validateOperatorResult(page));
});
test("metadata contract rejects omitted candidate preprocessing", () => {
  const page = metadata(); page.timing.paired[0].rows[1].dispatches.quantize = 0;
  assert.throws(() => validateOperatorResult(page), /preprocessing dispatch counts/);
});
test("metadata contract rejects an incomplete GPU coverage count", () => {
  const page = metadata(); page.fixtures[0].validation.words[0]--;
  assert.throws(() => validateOperatorResult(page), /GPU status\/coverage/);
});
test("metadata contract rejects a duration that disagrees with the raw timestamp span", () => {
  const page = metadata(); page.timing.paired[0].rows[1].timestampNs.elapsed = "1";
  assert.throws(() => validateOperatorResult(page), /timestamp span/);
});
test("metadata contract rejects completed results with an open map", () => {
  const page = metadata(); page.cleanup.mapsOpen = 1;
  assert.throws(() => validateOperatorResult(page));
});
test("metadata contract accepts a retained warning truncated by message length", () => {
  const page = metadata();
  page.compilation[0].diagnostics = { totalMessages: 1, totalErrors: 0, truncated: true,
    messages: [{ type: "warning", message: "w".repeat(256), messageTruncated: true }] };
  assert.equal(validateOperatorResult(page).performanceAccepted, false);
});
test("metadata contract rejects missing retained fixture input identities", () => {
  const page = metadata(); delete page.fixtures[2].inputIdentities;
  assert.throws(() => validateOperatorResult(page), /input byte count/);
});
test("metadata contract rejects a malformed fixture input digest", () => {
  const page = metadata(); page.fixtures[1].inputIdentities.S.sha256 = "missing";
  assert.throws(() => validateOperatorResult(page), /input SHA256/);
});
