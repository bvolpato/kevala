function sample(block, index) {
  const ticks = [], stages = [];
  let cursor = 1000n + BigInt(index) * 10000n, quantize = 0n, integerMatmul = 0n;
  for (let invocation = 0; invocation < 16; invocation++) {
    const qb = cursor, qe = qb + (invocation === 0 ? 0n : 10n), mb = qe + (invocation === 0 ? -2n : 3n), me = mb + 20n;
    ticks.push(qb, qe, mb, me); cursor = me + 4n;
    const stage = { invocation };
    for (const [name, offset, first, last] of [["quantize", invocation * 4, qb, qe], ["integerMatmul", invocation * 4 + 2, mb, me]]) {
      stage[name] = { beginIndex: offset, endIndex: offset + 1, begin: first.toString(), end: last.toString(), elapsed: (last - first).toString(), gpuMs: Number(last - first) / 1e6 };
    }
    stages.push(stage); quantize += qe - qb; integerMatmul += me - mb;
  }
  const total = ticks[63] - ticks[0], residual = total - quantize - integerMatmul;
  return { block, index, kernel: "candidate", invocations: 16, passes: 32,
    dispatches: { baseline: 0, quantize: 16, integerMatmul: 16, reduce: 0 }, rawTimestampNs: ticks.map(String),
    timestampNs: { begin: ticks[0].toString(), end: ticks[63].toString(), elapsed: total.toString() }, stages,
    interPassGapsNs: Array.from({ length: 31 }, (_, gap) => (ticks[(gap + 1) * 2] - ticks[gap * 2 + 1]).toString()),
    stageTotalsNs: { quantize: quantize.toString(), integerMatmul: integerMatmul.toString(), residual: residual.toString() },
    stageTotalsMs: { quantize: Number(quantize) / 1e6, integerMatmul: Number(integerMatmul) / 1e6, residual: Number(residual) / 1e6 },
    zeroDurationPasses: { quantize: 1, integerMatmul: 0 }, negativeInterPassGaps: 1,
    gpuMs: Number(total) / 1e6, perInvocationMs: Number(total) / 1e6 / 16, validation: validation() };
}

test("metadata preserves zero-duration passes and signed negative gaps without a ranking", () => {
  const result = validateSplitResult(metadata());
  assert.equal(result.diagnosticOnly, true); assert.equal(result.samples, 9);
  assert.equal(result.zeroDurationPasses.quantize, 9); assert.equal(result.negativeInterPassGaps, 9);
  assert.equal(result.performanceAccepted, undefined);
});
test("metadata rejects a clipped negative gap", () => {
  const page = metadata(); page.timing.samples[0].interPassGapsNs[0] = "0";
  assert.throws(() => validateSplitResult(page), /signed gap/);
});
test("metadata rejects an imputed positive duration for a raw zero pass", () => {
  const page = metadata(); page.timing.samples[0].stages[0].quantize.elapsed = "1";
  assert.throws(() => validateSplitResult(page));
});
test("metadata rejects incomplete raw64 timestamp coverage", () => {
  const page = metadata(); page.timing.samples[0].rawTimestampNs.pop();
  assert.throws(() => validateSplitResult(page), /raw64/);
});
test("metadata rejects a timed baseline dispatch in the candidate-only diagnostic", () => {
  const page = metadata(); page.timing.samples[0].dispatches.baseline = 1;
  assert.throws(() => validateSplitResult(page), /candidate-only/);
});
test("metadata rejects one omitted fixed sample", () => {
  const page = metadata(); page.timing.samples.pop();
  assert.throws(() => validateSplitResult(page), /sample count/);
});
test("metadata rejects missing fixture input identity", () => {
  const page = metadata(); delete page.fixtures[0].inputIdentities.X;
  assert.throws(() => validateSplitResult(page), /input byte count/);
});
test("metadata rejects incomplete actual GPU coverage", () => {
  const page = metadata(); page.timing.samples[1].validation.words[0]--;
  assert.throws(() => validateSplitResult(page), /GPU status/);
});
test("metadata rejects an acceptance decision on a diagnostic result", () => {
  const page = metadata(); page.timing.summary.accepted = true;
  assert.throws(() => validateSplitResult(page), /no acceptance/);
});
