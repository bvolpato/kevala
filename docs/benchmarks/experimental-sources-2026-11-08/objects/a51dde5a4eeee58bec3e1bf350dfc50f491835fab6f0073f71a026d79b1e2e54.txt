import ast
import hashlib
import json
import time
import unittest
from pathlib import Path


RUNNER = Path(__file__).resolve().parents[1] / "bench-gpu.py"
SOURCE_TREE = ast.parse(RUNNER.read_text(encoding="utf-8"), filename=str(RUNNER))
FUNCTIONS = {"poll_q8_page", "_q8_bounded_timeout_status", "_q8_timeout_console_errors", "_q8_browser_runner_failure_result", "thread_identity_delta", "_q8_check_original_anchors", "_q8_bounded_snapshot_partial"}
HELPER_TREE = ast.Module(body=[node for node in SOURCE_TREE.body if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)) and node.name in FUNCTIONS], type_ignores=[])
Q8_TERMINAL_EXPRESSION = "return Boolean(window.bench && window.bench.done === true);"
Q8_RESULT_EXPRESSION = "return window.bench || null;"
Q8_TIMEOUT_STATUS_EXPRESSION = "return (() => { const b = window.bench; if (!b || typeof b !== 'object') return { status: null, phase: null, error: null }; return { status: typeof b.status === 'string' ? b.status.slice(0, 64) : null, phase: typeof b.phase === 'string' ? b.phase.slice(0, 64) : null, error: typeof b.error === 'string' ? b.error.slice(0, 1024) : null }; })();"
NAMESPACE = {"hashlib": hashlib, "json": json, "time": time,
    "Q8_TERMINAL_EXPRESSION": Q8_TERMINAL_EXPRESSION,
    "Q8_RESULT_EXPRESSION": Q8_RESULT_EXPRESSION,
    "Q8_TIMEOUT_STATUS_EXPRESSION": Q8_TIMEOUT_STATUS_EXPRESSION,
    "Q8_TIMEOUT_STATUS_MAX_BYTES": 4096,
    "Q8_TIMEOUT_CONSOLE_MAX_BYTES": 16 * 1024,
    "Q8_TIMEOUT_CONSOLE_MAX_ERRORS": 8}
exec(compile(HELPER_TREE, str(RUNNER), "exec"), NAMESPACE)
poll_q8_page = NAMESPACE["poll_q8_page"]
_q8_bounded_timeout_status = NAMESPACE["_q8_bounded_timeout_status"]
_q8_timeout_console_errors = NAMESPACE["_q8_timeout_console_errors"]
_q8_browser_runner_failure_result = NAMESPACE["_q8_browser_runner_failure_result"]
thread_identity_delta = NAMESPACE["thread_identity_delta"]
_q8_bounded_snapshot_partial = NAMESPACE["_q8_bounded_snapshot_partial"]
_q8_check_original_anchors = NAMESPACE["_q8_check_original_anchors"]


class FakeBrowser:
    def __init__(self, terminal_values: list[bool], result: object, timeout_status: object = None, console_entries: list[dict] | None = None):
        self.terminal_values = iter(terminal_values)
        self.result = result
        self.timeout_status = timeout_status
        self.console_entries = console_entries or []
        self.calls: list[str] = []

    def execute_script(self, expression: str) -> object:
        self.calls.append(expression)
        if expression == Q8_TERMINAL_EXPRESSION:
            return next(self.terminal_values)
        if expression == Q8_RESULT_EXPRESSION:
            return self.result
        if expression == Q8_TIMEOUT_STATUS_EXPRESSION:
            return self.timeout_status
        raise AssertionError(f"unexpected script expression: {expression}")

    def get_log(self, log_type: str) -> list[dict]:
        if log_type != "browser":
            raise AssertionError(f"unexpected browser log type: {log_type}")
        return self.console_entries


class Q8TerminalPollTest(unittest.TestCase):
    def test_false_continues_then_true_breaks_and_fetches_full_result_once(self) -> None:
        browser = FakeBrowser([False, True], {"done": True, "q8SwiGluTrial": {"status": "complete"}})
        now = [0.0]
        samples: list[str] = []

        def wait(seconds: float) -> None:
            now[0] += seconds

        complete, result, timeout_status = poll_q8_page(browser.execute_script, deadline=1.0, check_diagnostics=samples.append,
            clock=lambda: now[0], wait=wait)

        self.assertTrue(complete)
        self.assertEqual(result, browser.result)
        self.assertIsNone(timeout_status)
        self.assertEqual(browser.calls, [
            Q8_TERMINAL_EXPRESSION,
            Q8_TERMINAL_EXPRESSION,
            Q8_RESULT_EXPRESSION,
        ])
        self.assertEqual(samples, ["poll", "poll", "terminal-before-result-fetch", "after-result-fetch"])
        self.assertEqual(browser.calls.count(Q8_RESULT_EXPRESSION), 1)

    def test_timeout_fetches_one_small_status_and_never_fetches_full_result(self) -> None:
        status = {"status": "running", "phase": "timing", "error": None}
        browser = FakeBrowser([False, False], {"done": True, "large": "x" * 100_000}, status)
        now = [0.0]

        def wait(seconds: float) -> None:
            now[0] += seconds

        complete, result, timeout_status = poll_q8_page(browser.execute_script, deadline=0.5,
            check_diagnostics=lambda _stage: None, clock=lambda: now[0], wait=wait)

        self.assertFalse(complete)
        self.assertIsNone(result)
        self.assertEqual(timeout_status, status)
        self.assertEqual(browser.calls, [Q8_TERMINAL_EXPRESSION, Q8_TERMINAL_EXPRESSION, Q8_TIMEOUT_STATUS_EXPRESSION])
        self.assertEqual(browser.calls.count(Q8_TIMEOUT_STATUS_EXPRESSION), 1)
        self.assertEqual(browser.calls.count(Q8_RESULT_EXPRESSION), 0)

    def test_timeout_console_summary_is_bounded_and_keeps_only_errors(self) -> None:
        browser = FakeBrowser([], None, console_entries=[
            {"level": "INFO", "message": "not retained"},
            *[{"level": "SEVERE", "timestamp": index, "source": "renderer", "message": "x" * 4_000} for index in range(12)],
        ])
        summary = _q8_timeout_console_errors(browser)
        self.assertEqual(summary["state"], "available")
        self.assertEqual(summary["errorCount"], 12)
        self.assertTrue(summary["truncated"])
        self.assertLessEqual(len(summary["errors"]), 8)
        self.assertTrue(all(len(entry["message"]) <= 256 for entry in summary["errors"]))
        self.assertLessEqual(len(json.dumps(summary, separators=(",", ":"), ensure_ascii=False).encode()), 16 * 1024)

        class UnavailableBrowserLog:
            def get_log(self, _log_type: str) -> list[dict]:
                raise RuntimeError("browser log endpoint unavailable")

        unavailable = _q8_timeout_console_errors(UnavailableBrowserLog())
        self.assertEqual(unavailable, {"state": "unavailable", "error": "browser log endpoint unavailable"})

    def test_timeout_status_capture_error_is_returned_without_raising(self) -> None:
        browser = FakeBrowser([False, False], None, {"status": "running"})
        now = [0.0]

        def fail_before_status(stage: str) -> None:
            if stage == "timeout-before-status-fetch":
                raise RuntimeError("diagnostic sample failed")

        complete, result, status = poll_q8_page(browser.execute_script, deadline=0.5,
            check_diagnostics=fail_before_status,
            clock=lambda: now[0], wait=lambda seconds: now.__setitem__(0, now[0] + seconds))
        self.assertFalse(complete)
        self.assertIsNone(result)
        self.assertEqual(status, {"statusReadError": "diagnostic sample failed"})
        self.assertEqual(browser.calls, [Q8_TERMINAL_EXPRESSION, Q8_TERMINAL_EXPRESSION])

    def test_timeout_status_fields_are_allowlisted_and_truncated(self) -> None:
        bounded = _q8_bounded_timeout_status({"status": "r" * 100, "phase": "timing", "error": "e" * 2_000, "result": "x" * 100_000})
        self.assertEqual(set(bounded), {"status", "phase", "error"})
        self.assertEqual(len(bounded["status"]), 64)
        self.assertEqual(len(bounded["error"]), 1024)

    def test_post_timeout_snapshot_error_preserves_page_and_console_diagnostics(self) -> None:
        previous = {"error": "q8 page did not expose its terminal flag before timeout",
            "timeoutDiagnostics": {"page": {"status": "running", "phase": "timing", "error": None},
                "browserConsole": {"state": "available", "errorCount": 1, "errors": [{"level": "SEVERE", "message": "gpu error"}]}}}
        result = _q8_browser_runner_failure_result(previous, RuntimeError("after-benchmark-result scope changed"))

        self.assertIn("after-benchmark-result scope changed", result["error"])
        self.assertIn("q8 page did not expose its terminal flag before timeout", result["error"])
        self.assertEqual(result["timeoutDiagnostics"]["page"], previous["timeoutDiagnostics"]["page"])
        self.assertEqual(result["timeoutDiagnostics"]["browserConsole"], previous["timeoutDiagnostics"]["browserConsole"])
        self.assertEqual(result["timeoutDiagnostics"]["postTimeoutRunnerError"], "after-benchmark-result scope changed")
        self.assertNotIn("postTimeoutRunnerError", previous["timeoutDiagnostics"])

    def test_browser_cleanup_error_preserves_successful_terminal_result(self) -> None:
        previous = {"status": "ok", "metricMs": [12.5, 12.25], "correctness": {"exact": True},
            "requestOutput": {"answer": "refund", "probability": 0.75}, "diagnostics": {"phase": "complete"}}
        result = _q8_browser_runner_failure_result(previous, RuntimeError("browser cleanup: quit failed"))

        self.assertEqual({key: result[key] for key in previous}, previous)
        self.assertEqual(result["browserRunnerError"], "browser cleanup: quit failed")
        self.assertEqual(result["error"], "browser runner: browser cleanup: quit failed")
        self.assertNotIn("error", previous)


class Q8ThreadSnapshotTest(unittest.TestCase):
    CGROUP = "0::/user.slice/kevala-q8.scope"

    def thread(self, tid: int, start_ticks: str = "500", *, cgroup: str | None = None, process_start: str = "400") -> dict:
        return {"pid": 41, "tid": tid, "startTicks": start_ticks, "processStartTicks": process_start,
            "processStartTicksAfter": process_start, "cgroup": cgroup or self.CGROUP, "cgroupAfter": cgroup or self.CGROUP}

    def test_individually_verified_new_tid_is_recorded_as_churn(self) -> None:
        delta = thread_identity_delta([self.thread(41), self.thread(42, "501")],
            [self.thread(41), self.thread(43, "502")], self.CGROUP)
        self.assertEqual([row["tid"] for row in delta["newThreads"]], [43])
        self.assertEqual([row["tid"] for row in delta["disappearedThreads"]], [42])
        self.assertEqual(delta["stableThreadCount"], 1)

    def test_escaped_cgroup_is_rejected(self) -> None:
        with self.assertRaisesRegex(ValueError, "escaped its exact cgroup"):
            thread_identity_delta([], [self.thread(43, cgroup="0::/outside")], self.CGROUP)

    def test_reused_pid_or_tid_identity_is_rejected(self) -> None:
        with self.assertRaisesRegex(ValueError, "TID 42 identity was reused"):
            thread_identity_delta([self.thread(42, "501")], [self.thread(42, "999")], self.CGROUP)
        with self.assertRaisesRegex(ValueError, "PID 41 identity was reused between TID inventories"):
            thread_identity_delta([self.thread(42, "501")],
                [self.thread(43, "502", process_start="401")], self.CGROUP)

    def test_absent_original_anchor_passes_and_same_identity_fails(self) -> None:
        phases = []
        original = {"driver": {"pid": 900, "startTicks": "100"}}

        def phase(name: str, **identity: object) -> None:
            phases.append((name, identity))

        def absent(_pid: int) -> dict:
            raise FileNotFoundError

        self.assertEqual(_q8_check_original_anchors(original, absent, phase), {"driver": {"state": "absent"}})
        self.assertEqual(phases[-1], ("check-original-anchor", {"anchorName": "driver", "pid": 900}))

        with self.assertRaisesRegex(ValueError, "driver process survived browser quit"):
            _q8_check_original_anchors(original, lambda _pid: {"pid": 900, "startTicks": "100"}, phase)

    def test_nested_churn_is_trimmed_under_the_partial_size_cap(self) -> None:
        churn = [self.thread(100 + index, f"{500 + index}") | {"detail": "x" * 280} for index in range(2048)]
        partial = {"stage": "regression", "roots": [41], "phases": [{"phase": "current-tid-inventory"}],
            "current": {"phase": "current-tid-inventory", "pid": 41, "tid": 100,
                "tidEvidence": {"before": self.thread(100), "after": self.thread(100)}},
            "assurance": "sequential samples", "error": "bounded synthetic overflow", "passes": [
                {"scan": 0, "threads": [], "liveThreads": [], "threadChurn": {
                    "newThreads": churn, "disappearedThreads": churn, "stableThreadCount": 0}}]}

        bounded = _q8_bounded_snapshot_partial(partial)

        self.assertLessEqual(len(json.dumps(bounded, separators=(",", ":"), ensure_ascii=False).encode()), 512 * 1024)
        churn_result = bounded["passes"][0]["threadChurn"]
        self.assertEqual(churn_result["newThreadsTotal"], 2048)
        self.assertTrue(churn_result["newThreadsTruncated"])
        self.assertEqual(bounded["current"]["tidEvidence"]["before"]["tid"], 100)


if __name__ == "__main__":
    unittest.main()
